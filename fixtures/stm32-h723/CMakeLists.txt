cmake_minimum_required(VERSION 3.22)
project(can_h723 C ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(STM32_CUBE_H7 "$ENV{HOME}/STM32Cube/Repository/STM32Cube_FW_H7_V1.13.0" CACHE PATH "STM32CubeH7 firmware root")
option(CAN_DUAL_BENCH "Enable independent FDCAN2 on PB12/PB13 for two-analyzer bench" OFF)
option(CAN_LOOPBACK "Use internal loopback instead of external CAN bus" OFF)
set(SIM_PROFILE "" CACHE STRING "Optional override: SIM_DJI_C620/C610/GM6020 or SIM_DM_MIT/POS_VEL/VEL/POS_FORCE or SIM_ALL_FEEDBACK")
if(NOT SIM_PROFILE STREQUAL "" AND NOT SIM_PROFILE MATCHES "^(SIM_DJI_(C620|C610|GM6020)|SIM_DM_(MIT|POS_VEL|VEL|POS_FORCE)|SIM_ALL_FEEDBACK|[1-8])$")
  message(FATAL_ERROR "Invalid SIM_PROFILE")
endif()
set(HAL_DIR "${STM32_CUBE_H7}/Drivers/STM32H7xx_HAL_Driver")
set(CMSIS_DIR "${STM32_CUBE_H7}/Drivers/CMSIS")
if(NOT EXISTS "${HAL_DIR}/Src/stm32h7xx_hal_fdcan.c")
  message(FATAL_ERROR "Set STM32_CUBE_H7 to an installed STM32CubeH7 package")
endif()
add_executable(can_h723 Core/Src/main.c Core/Src/can_bus.c Core/Src/lcd_display.c Core/Src/motor_sim.c Core/Src/imu_sim.c startup_stm32h723xx.s
  "${CMSIS_DIR}/Device/ST/STM32H7xx/Source/Templates/system_stm32h7xx.c")
foreach(driver hal hal_cortex hal_rcc hal_rcc_ex hal_gpio hal_pwr hal_pwr_ex hal_flash hal_flash_ex hal_fdcan hal_spi hal_spi_ex hal_dma)
  target_sources(can_h723 PRIVATE "${HAL_DIR}/Src/stm32h7xx_${driver}.c")
endforeach()
target_include_directories(can_h723 PRIVATE Core/Inc "${HAL_DIR}/Inc" "${CMSIS_DIR}/Include" "${CMSIS_DIR}/Device/ST/STM32H7xx/Include")
target_compile_definitions(can_h723 PRIVATE USE_HAL_DRIVER STM32H723xx USE_PWR_LDO_SUPPLY CAN_LOOPBACK=$<BOOL:${CAN_LOOPBACK}> CAN_DUAL_BENCH=$<BOOL:${CAN_DUAL_BENCH}>)
if(NOT SIM_PROFILE STREQUAL "")
  target_compile_definitions(can_h723 PRIVATE SIM_PROFILE=${SIM_PROFILE})
endif()
target_compile_options(can_h723 PRIVATE -mcpu=cortex-m7 -mthumb -mfpu=fpv5-d16 -mfloat-abi=hard -Wall -Wextra -ffunction-sections -fdata-sections -Og -g3)
target_link_options(can_h723 PRIVATE -mcpu=cortex-m7 -mthumb -mfpu=fpv5-d16 -mfloat-abi=hard --specs=nano.specs --specs=nosys.specs
  "-T${CMAKE_CURRENT_SOURCE_DIR}/STM32H723xG_flash.ld" -Wl,--gc-sections -Wl,-Map=can_h723.map -Wl,--print-memory-usage)
target_link_libraries(can_h723 PRIVATE m)
set_target_properties(can_h723 PROPERTIES SUFFIX .elf LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/STM32H723xG_flash.ld")
add_custom_command(TARGET can_h723 POST_BUILD
  COMMAND arm-none-eabi-objcopy -O ihex $<TARGET_FILE:can_h723> can_h723.hex
  COMMAND arm-none-eabi-objcopy -O binary $<TARGET_FILE:can_h723> can_h723.bin)
